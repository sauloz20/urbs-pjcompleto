<?php

/**
 * Urbs - Conexão com o banco de dados
 * Utilizando PDO e MySQL
*/

$host = getenv('DB_HOST') ?: 'localhost';
$database = getenv('DB_NAME') ?: 'urbs';
$username = getenv('DB_USER') ?: 'root';
$password = getenv('DB_PASS') ?: '';

try {
    $pdo = new PDO(
        "mysql:host={$host};dbname={$database};charset=utf8mb4",
        $username,
        $password,
        [
            PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES => false
        ]
    );

} catch (PDOException $e) {
    error_log('Erro de conexão Urbs: ' . $e->getMessage());

    http_response_code(500);
    exit('Não foi possível conectar ao banco de dados.');
}

<?php
// Iniciar a sessão. Isso precisa ser a primeira coisa antes de qualquer saída de dados.
session_start();

header('Content-Type: application/json; charset=utf-8');

require_once __DIR__ . '/../../../config/database.php';

function response(int $status, array $data): void
{
    http_response_code($status);
    echo json_encode($data, JSON_UNESCAPED_UNICODE);
    exit;
}

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    response(405, [
        'success' => false,
        'message' => 'Método não permitido. Utilize POST.'
    ]);
}

$input = json_decode(file_get_contents('php://input'), true);

if (!is_array($input)) {
    response(400, [
        'success' => false,
        'message' => 'Dados inválidos ou formato JSON incorreto.'
    ]);
}

$email = strtolower(trim($input['email'] ?? ''));
$password = $input['password'] ?? '';

if ($email === '' || $password === '') {
    response(400, [
        'success' => false,
        'message' => 'Informe o e-mail e a senha.'
    ]);
}

try {
    // Buscar o usuário pelo e-mail, incluindo a categoria e empresa (se for resolver)
    $sql = "SELECT id, name, password, role, category_id, company_name 
            FROM users 
            WHERE email = :email 
            LIMIT 1";
    $stmt = $pdo->prepare($sql);
    $stmt->execute([':email' => $email]);
    $user = $stmt->fetch(PDO::FETCH_ASSOC);

    if (!$user || !password_verify($password, $user['password'])) {
        response(401, [
            'success' => false,
            'message' => 'E-mail ou senha incorretos.'
        ]);
    }

    // Renovar o ID da sessão após autenticar para reduzir risco de session fixation.
    session_regenerate_id(true);

    // Salvar dados na sessão
    $_SESSION['user_id'] = (int) $user['id'];
    $_SESSION['role'] = $user['role'];
    $_SESSION['name'] = $user['name'];
    $_SESSION['category_id'] = $user['category_id'] ? (int) $user['category_id'] : null;

    response(200, [
        'success' => true,
        'message' => 'Login realizado com sucesso!',
        'user' => [
            'id' => (int) $user['id'],
            'name' => $user['name'],
            'role' => $user['role'],
            'category_id' => $_SESSION['category_id'],
            'company_name' => $user['company_name']
        ]
    ]);

} catch (PDOException $e) {
    error_log('Erro no login Urbs: ' . $e->getMessage());
    response(500, [
        'success' => false,
        'message' => 'Não foi possível realizar o login devido a um erro no servidor.'
    ]);
}

<?php

session_start();

header('Content-Type: application/json; charset=utf-8');

require_once __DIR__ . '/../../../config/database.php';

// Função para padronizar as respostas da API
function response(int $status, array $data): void
{
    http_response_code($status);

    echo json_encode(
        $data,
        JSON_UNESCAPED_UNICODE
    );

    exit;
}

// Aceitar somente requisições POST
if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    response(405, [
        'success' => false,
        'message' => 'Método não permitido. Utilize POST.'
    ]);
}

// Receber os dados enviados em JSON
$input = json_decode(
    file_get_contents('php://input'),
    true
);

// Verificar se o JSON é válido
if (!is_array($input)) {
    response(400, [
        'success' => false,
        'message' => 'Dados inválidos ou formato JSON incorreto.'
    ]);
}

// Capturar e tratar os campos
$name = trim($input['name'] ?? '');
$email = strtolower(trim($input['email'] ?? ''));
$password = $input['password'] ?? '';

// Validar campos obrigatórios
if ($name === '' || $email === '' || $password === '') {
    response(400, [
        'success' => false,
        'message' => 'Preencha todos os campos obrigatórios.'
    ]);
}

// Validar tamanho do nome
$nameLength = function_exists('mb_strlen')
    ? mb_strlen($name, 'UTF-8')
    : strlen($name);

if ($nameLength > 120) {
    response(400, [
        'success' => false,
        'message' => 'O nome deve possuir no máximo 120 caracteres.'
    ]);
}

// Validar e-mail
if (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
    response(400, [
        'success' => false,
        'message' => 'Informe um endereço de e-mail válido.'
    ]);
}

// Validar tamanho do e-mail
if (strlen($email) > 150) {
    response(400, [
        'success' => false,
        'message' => 'O e-mail deve possuir no máximo 150 caracteres.'
    ]);
}

// Validar senha
if (strlen($password) < 8) {
    response(400, [
        'success' => false,
        'message' => 'A senha deve possuir pelo menos 8 caracteres.'
    ]);
}

try {

    // Gerar hash seguro da senha
    $passwordHash = password_hash(
        $password,
        PASSWORD_DEFAULT
    );

    // Preparar a inserção no banco
    $sql = "INSERT INTO users (name, email, password)
            VALUES (:name, :email, :password)";

    $stmt = $pdo->prepare($sql);

    // Executar a consulta
    $stmt->execute([
        ':name' => $name,
        ':email' => $email,
        ':password' => $passwordHash
    ]);

    $userId = (int) $pdo->lastInsertId();
    session_regenerate_id(true);
    $_SESSION['user_id'] = $userId;
    $_SESSION['role'] = 'citizen';
    $_SESSION['name'] = $name;
    $_SESSION['category_id'] = null;

    // Retornar sucesso
    response(201, [
        'success' => true,
        'message' => 'Usuário cadastrado com sucesso!',
        'user' => [
            'id' => $userId,
            'name' => $name,
            'email' => $email,
            'role' => 'citizen'
        ]
    ]);

} catch (PDOException $e) {

    // Verificar se o e-mail já está cadastrado
    if ($e->getCode() === '23000') {
        response(409, [
            'success' => false,
            'message' => 'Este e-mail já está cadastrado.'
        ]);
    }

    // Registrar erro internamente
    error_log('Erro no cadastro CitySense: ' . $e->getMessage());

    response(500, [
        'success' => false,
        'message' => 'Não foi possível realizar o cadastro.'
    ]);
}

<?php
// backend/public/api/occurrences/create.php

// 1. Carregar middleware e banco de dados
require_once __DIR__ . '/../../../src/middlewares/auth.php';
require_once __DIR__ . '/../../../config/database.php';

// 2. Exigir autenticação antes de qualquer coisa
requireLogin();

header('Content-Type: application/json; charset=utf-8');

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    http_response_code(405);
    echo json_encode(['success' => false, 'message' => 'Método não permitido.']);
    exit;
}

$input = json_decode(file_get_contents('php://input'), true);
if (!is_array($input)) {
    http_response_code(400);
    echo json_encode(['success' => false, 'message' => 'Envie os dados da ocorrência em formato JSON válido.'], JSON_UNESCAPED_UNICODE);
    exit;
}

// Capturar os dados
$title = trim($input['title'] ?? '');
$description = trim($input['description'] ?? '');
$address = trim($input['address'] ?? '');
$category_id = filter_var($input['category_id'] ?? 0, FILTER_VALIDATE_INT);
$latitude = $input['latitude'] ?? null;
$longitude = $input['longitude'] ?? null;

// Validação básica
if ($title === '' || $description === '' || !$category_id || $latitude === null || $longitude === null) {
    http_response_code(400);
    echo json_encode(['success' => false, 'message' => 'Preencha todos os campos obrigatórios e a localização.']);
    exit;
}

if (mb_strlen($title, 'UTF-8') > 150 || mb_strlen($address, 'UTF-8') > 180 || mb_strlen($description, 'UTF-8') > 5000) {
    http_response_code(400);
    echo json_encode(['success' => false, 'message' => 'Um dos campos excede o tamanho permitido.'], JSON_UNESCAPED_UNICODE);
    exit;
}

$latitude = filter_var($latitude, FILTER_VALIDATE_FLOAT);
$longitude = filter_var($longitude, FILTER_VALIDATE_FLOAT);
if ($latitude === false || $longitude === false || $latitude < -90 || $latitude > 90 || $longitude < -180 || $longitude > 180) {
    http_response_code(400);
    echo json_encode(['success' => false, 'message' => 'As coordenadas informadas são inválidas.'], JSON_UNESCAPED_UNICODE);
    exit;
}

try {
    // Iniciar a transação (garante que ou tudo salva, ou nada salva)
    $pdo->beginTransaction();

    // 1. Inserir a ocorrência
    $sqlOccurrence = "INSERT INTO occurrences (user_id, category_id, title, description, address, latitude, longitude, status)
                      VALUES (:user_id, :category_id, :title, :description, :address, :latitude, :longitude, 'pending')";
    
    $stmt = $pdo->prepare($sqlOccurrence);
    $stmt->execute([
        ':user_id'     => $_SESSION['user_id'], // Pego direto da sessão segura!
        ':category_id' => $category_id,
        ':title'       => $title,
        ':description' => $description,
        ':address'     => $address,
        ':latitude'    => $latitude,
        ':longitude'   => $longitude
    ]);

    $occurrence_id = $pdo->lastInsertId();

    // 2. Inserir o registro no histórico inicial
    $sqlHistory = "INSERT INTO occurrence_history (occurrence_id, user_id, new_status, comment) 
                   VALUES (:occurrence_id, :user_id, 'pending', 'Ocorrência registrada pelo cidadão.')";
    
    $stmtHistory = $pdo->prepare($sqlHistory);
    $stmtHistory->execute([
        ':occurrence_id' => $occurrence_id,
        ':user_id'       => $_SESSION['user_id']
    ]);

    // Confirmar a transação
    $pdo->commit();

    http_response_code(201);
    echo json_encode([
        'success' => true,
        'message' => 'Ocorrência registrada com sucesso!',
        'occurrence_id' => $occurrence_id
    ], JSON_UNESCAPED_UNICODE);

} catch (PDOException $e) {
    // Desfazer tudo caso ocorra um erro no meio do processo
    if ($pdo->inTransaction()) {
        $pdo->rollBack();
    }
    
    http_response_code(500);
    echo json_encode([
        'success' => false,
        'message' => 'Erro interno ao registrar a ocorrência.'
    ], JSON_UNESCAPED_UNICODE);
}

<?php
// backend/public/api/occurrences/delete.php

require_once __DIR__ . '/../../../src/middlewares/auth.php';
require_once __DIR__ . '/../../../config/database.php';

requireLogin();

header('Content-Type: application/json; charset=utf-8');

if (!in_array($_SERVER['REQUEST_METHOD'], ['POST', 'DELETE'], true)) {
    http_response_code(405);
    echo json_encode(['success' => false, 'message' => 'Método não permitido. Utilize DELETE ou POST.']);
    exit;
}

$input = json_decode(file_get_contents('php://input'), true);
if (!is_array($input)) {
    http_response_code(400);
    echo json_encode(['success' => false, 'message' => 'Envie os dados em formato JSON válido.'], JSON_UNESCAPED_UNICODE);
    exit;
}
$occurrence_id = filter_var($input['occurrence_id'] ?? 0, FILTER_VALIDATE_INT);

if (!$occurrence_id) {
    http_response_code(400);
    echo json_encode(['success' => false, 'message' => 'Informe o ID da ocorrência para exclusão.']);
    exit;
}

try {
    // 1. Verificar se a ocorrência existe e quem é o dono
    $stmtCheck = $pdo->prepare("SELECT user_id, status FROM occurrences WHERE id = :id LIMIT 1");
    $stmtCheck->execute([':id' => $occurrence_id]);
    $occurrence = $stmtCheck->fetch(PDO::FETCH_ASSOC);

    if (!$occurrence) {
        http_response_code(404);
        echo json_encode(['success' => false, 'message' => 'Ocorrência não encontrada.']);
        exit;
    }

    // 2. Regra de negócio de autorização:
    // Master pode apagar qualquer uma. Cidadão só apaga a própria se ainda estiver 'pending'.
    $isMaster = ($_SESSION['role'] === 'master');
    $isOwnerPending = ((int) $occurrence['user_id'] === (int) $_SESSION['user_id'] && $occurrence['status'] === 'pending');

    if (!$isMaster && !$isOwnerPending) {
        http_response_code(403);
        echo json_encode([
            'success' => false,
            'message' => 'Sem permissão para excluir. Cidadãos só podem excluir ocorrências próprias ainda pendentes.'
        ], JSON_UNESCAPED_UNICODE);
        exit;
    }

    // 3. Transação para remover o histórico vinculado e depois a ocorrência (evita erro de Foreign Key)
    $pdo->beginTransaction();

    $stmtDelHistory = $pdo->prepare("DELETE FROM occurrence_history WHERE occurrence_id = :id");
    $stmtDelHistory->execute([':id' => $occurrence_id]);

    $stmtDelOcc = $pdo->prepare("DELETE FROM occurrences WHERE id = :id");
    $stmtDelOcc->execute([':id' => $occurrence_id]);

    $pdo->commit();

    http_response_code(200);
    echo json_encode([
        'success' => true,
        'message' => 'Ocorrência excluída com sucesso!'
    ], JSON_UNESCAPED_UNICODE);

} catch (PDOException $e) {
    if ($pdo->inTransaction()) {
        $pdo->rollBack();
    }
    error_log('Erro ao excluir ocorrência no Urbs: ' . $e->getMessage());
    http_response_code(500);
    echo json_encode(['success' => false, 'message' => 'Erro interno ao excluir a ocorrência.']);
}

    exit;

try {
    $conditions = [];
    $params = [];

    // Filtro opcional por status (?status=pending)
    if (!empty($_GET['status'])) {
        $conditions[] = "o.status = :status";
        $params[':status'] = $_GET['status'];
    }

    // Se for um 'resolver' e pedir visão restrita da empresa (?my_company=1), força a categoria dele
    if ($_SESSION['role'] === 'resolver' && !empty($_GET['my_company']) && $_SESSION['category_id'] !== null) {
        $conditions[] = "o.category_id = :category_id";
        $params[':category_id'] = $_SESSION['category_id'];
    } elseif (!empty($_GET['category_id'])) {
        // Filtro manual por categoria (?category_id=1)
        $conditions[] = "o.category_id = :category_id";
        $params[':category_id'] = (int) $_GET['category_id'];
    }

    // Filtro opcional para o cidadão ver apenas as próprias ocorrências (?my_occurrences=1)
    if (!empty($_GET['my_occurrences'])) {
        $conditions[] = "o.user_id = :user_id";
        $params[':user_id'] = $_SESSION['user_id'];
    }

    $whereSql = count($conditions) > 0 ? 'WHERE ' . implode(' AND ', $conditions) : '';

    $sql = "SELECT o.id, o.user_id, o.title, o.description, o.address, o.latitude, o.longitude, o.status, o.created_at,
                   o.category_id, c.name AS category_name, u.name AS user_name
            FROM occurrences o
            JOIN categories c ON o.category_id = c.id
            JOIN users u ON o.user_id = u.id
            $whereSql
            ORDER BY o.created_at DESC";
            
    $stmt = $pdo->prepare($sql);
    $stmt->execute($params);
    $occurrences = $stmt->fetchAll(PDO::FETCH_ASSOC);

    http_response_code(200);

<?php
// backend/public/api/occurrences/stats.php

require_once __DIR__ . '/../../../src/middlewares/auth.php';
require_once __DIR__ . '/../../../config/database.php';

requireLogin();

header('Content-Type: application/json; charset=utf-8');

try {
    $where = "";
    $params = [];

    // Se for empresa solucionadora, filtra os KPIs apenas para a categoria que ela atende
    if ($_SESSION['role'] === 'resolver' && $_SESSION['category_id'] !== null) {
        $where = "WHERE o.category_id = :category_id";
        $params[':category_id'] = $_SESSION['category_id'];
    }

    // 1. Contagem por status
    $sqlStatus = "SELECT status, COUNT(*) AS total FROM occurrences o $where GROUP BY status";
    $stmtStatus = $pdo->prepare($sqlStatus);
    $stmtStatus->execute($params);
    $statusRows = $stmtStatus->fetchAll(PDO::FETCH_ASSOC);

    $counts = [
        'total'       => 0,
        'pending'     => 0,
        'in_analysis' => 0,
        'in_progress' => 0,
        'resolved'    => 0,
        'rejected'    => 0
    ];

    foreach ($statusRows as $row) {
        $st = $row['status'];
        $qty = (int) $row['total'];
        if (isset($counts[$st])) {
            $counts[$st] = $qty;
        }
        $counts['total'] += $qty;
    }

    // 2. Contagem por categoria
    $sqlCat = "SELECT c.name AS category_name, COUNT(o.id) AS total
               FROM categories c
               LEFT JOIN occurrences o ON c.id = o.category_id
               " . ($where ? "WHERE c.id = :category_id" : "") . "
               GROUP BY c.id, c.name
               ORDER BY total DESC";
